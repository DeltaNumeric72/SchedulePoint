=== PROBE F — CP-SAT assumption support (SPEC-04 §5 tier T1) ===
Model: three assumption literals, a0 => x0=1, a1 => x0=0, a2 => x1=1. a0/a1 conflict.
from ortools.sat.python import cp_model
m = cp_model.CpModel()
xs = [m.NewBoolVar("x%d" % i) for i in range(3)]
a = [m.NewBoolVar("a%d" % i) for i in range(3)]
# a0 -> x0 ; a1 -> not x0 ; a2 -> x1  => a0,a1 conflict
m.Add(xs[0] == 1).OnlyEnforceIf(a[0])
m.Add(xs[0] == 0).OnlyEnforceIf(a[1])
m.Add(xs[1] == 1).OnlyEnforceIf(a[2])
m.AddAssumptions(a)
s = cp_model.CpSolver()
s.parameters.num_search_workers = 1
s.parameters.interleave_search = True
s.parameters.max_deterministic_time = 5.0
s.parameters.random_seed = 7
st = s.Solve(m)
print("status", s.StatusName(st))
print("core", list(s.SufficientAssumptionsForInfeasibility()))
print("indices of a", [v.Index() for v in a])
print("ortools", __import__("ortools").__spec__ is not None)
import ortools
from importlib import metadata
print("ver", metadata.version("ortools"))
